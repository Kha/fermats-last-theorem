Hopf towers, Hopf subalgebras, quotient torsors, translation automorphisms

For commutative Hopf k-algebras A, C with C an A-algebra in a scalar tower over k, the class IsHopfTower k A C records two equations: the comultiplication of \mathrm{algebraMap}_{A\to C}(a) is the image of \mathrm{comul}(a) under the tensor square of the structure map, and counits are compatible. From this the structure map is packaged as a bialgebra homomorphism toBialgHom, antipodes commute with it, Sweedler representations Coalgebra.Repr transport along it (reprMap), and towers compose (trans), with C over itself as the reflexive case. augIdeal k A is the kernel of the counit algebra map, with the standard facts that a-\varepsilon(a) lies in it and that it is proper over a nontrivial base. IsHopfSubalgebra k K, for a subalgebra K of a Hopf algebra H, asserts that \mathrm{comul}(x) lies in the k-span of \{a\otimes b: a,b\in K\} for x\in K and that K is stable under the antipode; assuming H and K flat over k, so that K\otimes_k K\to H\otimes_k H is injective, the comultiplication, counit and antipode restrict and equip K with coalgebra, bialgebra and Hopf algebra structures for which k\to K\to H is a Hopf tower. In the HopfTower namespace the antipode becomes an algebra endomorphism antipodeAlgHom, convolution-inverse to the identity on points. For a tower k\to A\to B\to C of Hopf algebras, augIdealMap k A B is the ideal A^{+}B generated by the image of the augmentation ideal; the maps b\mapsto\sum b_{(1)}\otimes[b_{(2)}] and [b]\mapsto\sum S(b_{(1)})\otimes b_{(2)} are shown mutually inverse, yielding the C-algebra isomorphism galoisEquiv between C\otimes_A B and C\otimes_k(B/A^{+}B), natural in B. Finally, for \varphi:C\to k, translate is x\mapsto\sum\varphi(x_{(2)})\,x_{(1)}: an algebra automorphism translateEquiv with \varepsilon\circ\tau_\varphi=\varphi, composition matching convolution product, pulling the augmentation ideal back to \ker\varphi, and compatible with tower structure maps.

Relation to Mathlib

Built on Mathlib's Coalgebra, Bialgebra, HopfAlgebra, Coalgebra.Repr and the convolution monoid WithConv; the notions of Hopf tower, Hopf subalgebra with its induced Hopf structure, augmentation ideal, the quotient-torsor isomorphism and the translation automorphisms are the project's own.

Where it is used

This is the Hopf-algebraic (affine group scheme) infrastructure on which the treatment of quotients of commutative Hopf algebras and of flatness over Hopf subalgebras rests, in the style of Takeuchi's theorem.

theorem galoisInv_fwdAux (b : B) : galoisInv k A B C (fwdAux k A B C b) = 1 ⊗ₜ[A] b := by
  let 𝓡 := ℛ k b
  let a₁ : (i : _) → Coalgebra.Repr k (𝓡.left i) (_ × _) := fun i => ℛ k (𝓡.left i)
  let a₂ : (i : _) → Coalgebra.Repr k (𝓡.right i) (_ × _) := fun i => ℛ k (𝓡.right i)
  have key := congr(θ₁ k A B C $(sum_tmul_tmul_eq 𝓡 a₁ a₂))
  simp only [map_sum, θ₁_tmul] at key

  have lhs : galoisInv k A B C (fwdAux k A B C b) = ∑ i ∈ 𝓡.index, ∑ j ∈ (a₂ i).index,
      (algebraMap B C (𝓡.left i) * antipode k (algebraMap B C ((a₂ i).left j))) ⊗ₜ[A]
        (a₂ i).right j := by
    rw [fwdAux_apply k A B C 𝓡, map_sum]
    refine Finset.sum_congr rfl fun i _ => ?_
    rw [galoisInv_tmul, invQuot_mk, invAux_apply k A B C (a₂ i), Finset.mul_sum]
    refine Finset.sum_congr rfl fun j _ => ?_
    rw [Algebra.TensorProduct.tmul_mul_tmul, one_mul]
  have rhs : ∑ i ∈ 𝓡.index, ∑ j ∈ (a₁ i).index,
      (algebraMap B C ((a₁ i).left j) * antipode k (algebraMap B C ((a₁ i).right j))) ⊗ₜ[A]
        𝓡.right i = 1 ⊗ₜ[A] b := by
    have h : ∀ i ∈ 𝓡.index, ∑ j ∈ (a₁ i).index,
        (algebraMap B C ((a₁ i).left j) * antipode k (algebraMap B C ((a₁ i).right j))) ⊗ₜ[A]
          𝓡.right i = (1 : C) ⊗ₜ[A] (counit (R := k) (𝓡.left i) • 𝓡.right i) := fun i _ => by
      have h2 : ∑ j ∈ (a₁ i).index,
          algebraMap B C ((a₁ i).left j) * antipode k (algebraMap B C ((a₁ i).right j)) =
            algebraMap k C (counit (R := k) (algebraMap B C (𝓡.left i))) :=
        sum_mul_antipode_eq_algebraMap_counit (IsHopfTower.reprMap k B C (a₁ i))
      rw [← TensorProduct.sum_tmul, h2, IsHopfTower.counit_algebraMap,
        Algebra.algebraMap_eq_smul_one, ← TensorProduct.smul_tmul]
    rw [Finset.sum_congr rfl h, ← TensorProduct.tmul_sum, sum_counit_smul]
  rw [lhs, ← key, rhs]

  have lhs : galoisFwd k A B C (invAux k A B C b) = ∑ i ∈ 𝓡.index, ∑ j ∈ (a₂ i).index,

        Ideal.Quotient.mk (augIdealMap k A B) ((a₂ i).right j) := by
    rw [invAux_apply k A B C 𝓡, map_sum]
    refine Finset.sum_congr rfl fun i _ => ?_
    rw [galoisFwd_tmul, fwdAux_apply k A B C (a₂ i), Finset.mul_sum]
    refine Finset.sum_congr rfl fun j _ => ?_
    rw [Algebra.TensorProduct.tmul_mul_tmul, one_mul]
  have rhs : ∑ i ∈ 𝓡.index, ∑ j ∈ (a₁ i).index,
      (antipode k (algebraMap B C ((a₁ i).left j)) * algebraMap B C ((a₁ i).right j)) ⊗ₜ[k]
        Ideal.Quotient.mk (augIdealMap k A B) (𝓡.right i) =

theorem translate_translate (φ ψ : C →ₐ[k] k) (x : C) :
    translate k C φ (translate k C ψ x) = translate k C (ofConv (toConv φ * toConv ψ)) x := by
  let 𝓡 := ℛ k x
  let a₁ : (i : _) → Coalgebra.Repr k (𝓡.left i) (_ × _) := fun i => ℛ k (𝓡.left i)
  let a₂ : (i : _) → Coalgebra.Repr k (𝓡.right i) (_ × _) := fun i => ℛ k (𝓡.right i)

  let β : C ⊗[k] C →ₗ[k] k := LinearMap.mul' k k ∘ₗ TensorProduct.map φ.toLinearMap ψ.toLinearMap
  let θ : C ⊗[k] (C ⊗[k] C) →ₗ[k] C :=
    TensorProduct.lift ((LinearMap.lcomp k C β) ∘ₗ (LinearMap.lsmul k C).flip)
  have hθ : ∀ u v w : C, θ (u ⊗ₜ[k] (v ⊗ₜ[k] w)) = (φ v * ψ w) • u := fun u v w => by
    simp [θ, β]
  have key := congr(θ $(sum_tmul_tmul_eq 𝓡 a₁ a₂))
  simp only [map_sum, hθ] at key
  have lhs : translate k C φ (translate k C ψ x) =
      ∑ i ∈ 𝓡.index, ∑ j ∈ (a₁ i).index, (φ ((a₁ i).right j) * ψ (𝓡.right i)) • (a₁ i).left j := by
    rw [translate_apply ψ 𝓡, map_sum]
    refine Finset.sum_congr rfl fun i _ => ?_
    rw [map_smul, translate_apply φ (a₁ i), Finset.smul_sum]
    refine Finset.sum_congr rfl fun j _ => ?_
    rw [smul_smul, mul_comm]
  have rhs : translate k C (ofConv (toConv φ * toConv ψ)) x =
      ∑ i ∈ 𝓡.index, ∑ j ∈ (a₂ i).index, (φ ((a₂ i).left j) * ψ ((a₂ i).right j)) • 𝓡.left i := by
    rw [translate_apply _ 𝓡]
    refine Finset.sum_congr rfl fun i _ => ?_
    rw [convMul_apply' φ ψ (a₂ i), Finset.sum_smul]
  rw [lhs, key, ← rhs]
