definition module

Structure constants for a matrix order and its coordinates

Fix a family T : G \to M_n(\mathbb Z), a finite family u : \mathrm{Fin}\,K \to M_n(\mathbb Z), and integers c_{klm}, c^{(1)}_m, c^{(g)}_m. For a commutative ring R, uC and TC denote the entrywise images \bar u_k, \bar T_g \in M_n(R) under \mathbb Z \to R. Three constructions encode coordinates relative to u: elemM u R a, equal to \sum_k a_k \bar u_k for a \in R^K; cstar c R a b, the bilinear product on R^K given by (a \star b)_m = \sum_{k,l} a_k b_l c_{klm}; and Lm c R a, the K \times K matrix with entries (L_a)_{ml} = \sum_k a_k c_{klm}, whose action on vectors is L_a b = a \star b. The abbreviation AR T R is the R-subalgebra R[\bar T_g : g \in G] of M_n(R) generated by the image of TC.

The accompanying lemmas fall into groups: bilinearity of \star and additivity of elemM; compatibility of uC, TC, elemM, cstar, Lm with a ring homomorphism R \to R''; and, under the hypotheses u_k u_l = \sum_m c_{klm} u_m, \sum_m c^{(1)}_m u_m = 1, \sum_m c^{(g)}_m u_m = T_g over \mathbb Z, the corresponding identities over R together with \mathrm{elemM}(a \star b) = \mathrm{elemM}(a)\,\mathrm{elemM}(b). Two descent statements show that R-linear independence of the \bar u_k forces \mathbb Z-linear independence of the u_k, and that commutation of the \bar u_k over a characteristic-zero R forces commutation of the u_k; a further lemma derives commutation of the \bar u_k from commutation of the \bar T_g when each \bar u_k lies in R[\bar T_g]. Finally, assuming the \bar u_k are R-linearly independent with R-span equal to R[\bar T_g : g], uA lifts them into that subalgebra, basisA is the resulting R-basis, and crd is the associated coordinate map R[\bar T_g : g] \to R^K, shown to be inverse to elemM, to carry multiplication to \star, and to send 1, \bar T_g and \bar u_k to c^{(1)}, c^{(g)} and the standard basis vector.

Relation to Mathlib

Mathlib has no notion of a presentation of a matrix subring by structure constants; these are the project's own definitions, built on Mathlib's Matrix.map, Algebra.adjoin and Module.Basis.

Where it is used

The material provides explicit integral coordinates on the subring of M_n generated by a given family of integer matrices, so that properties of the base change R[\bar T_g] (linear independence, commutativity, products expressed through the structure constants) can be transferred between \mathbb Z and an arbitrary commutative ring.

def elemM (a : Fin K → R') : Matrix (Fin n) (Fin n) R' := ∑ k, a k • uC u R' k

theorem Lm_mulVec (a b : Fin K → R') : Lm c R' a *ᵥ b = cstar c R' a b := by
  funext m
  simp only [Lm, cstar, Matrix.mulVec, dotProduct, Matrix.of_apply, Finset.sum_mul]
  rw [Finset.sum_comm]
  refine Finset.sum_congr rfl fun k _ => Finset.sum_congr rfl fun l _ => ?_
  ring

theorem elemM_cstar (a b : Fin K → R') : elemM u R' (cstar c R' a b) = elemM u R' a * elemM u R' b := by
  have hR : elemM u R' a * elemM u R' b = ∑ k, ∑ l, ∑ m, (a k * b l * (c k l m : R')) • uC u R' m := by
    simp only [elemM, Finset.sum_mul, Finset.mul_sum, smul_mul_smul_comm, uC_mul u c R' hu_mul,
      Finset.smul_sum, smul_smul]
    exact Finset.sum_comm
  have hL : elemM u R' (cstar c R' a b) = ∑ m, ∑ k, ∑ l, (a k * b l * (c k l m : R')) • uC u R' m := by
    simp only [elemM, cstar]
    refine Finset.sum_congr rfl fun m _ => ?_
    rw [Finset.sum_smul]
    refine Finset.sum_congr rfl fun k _ => ?_
    rw [Finset.sum_smul]
  rw [hL, hR, Finset.sum_comm]
  refine Finset.sum_congr rfl fun k _ => ?_
  rw [Finset.sum_comm]

theorem mul_elemM_mul_eq_of_single {m' : Type*} [Fintype m'] (β : Matrix m' (Fin n) R')
    (e : Matrix (Fin n) (Fin n) R') (P : Matrix (Fin K) (Fin K) R' → Matrix m' (Fin n) R')
    (hP : ∀ κ, β * uC u R' κ * e = P (Lm c R' (Pi.single κ 1)))
    (hPadd : ∀ X Y, P (X + Y) = P X + P Y) (hPsmul : ∀ (z : R') X, P (z • X) = z • P X) (hP0 : P 0 = 0)
    (a : Fin K → R') : β * elemM u R' a * e = P (Lm c R' a) := by
  classical
  have h1 : β * elemM u R' a * e = ∑ κ, a κ • (β * uC u R' κ * e) := by
    simp only [elemM, Matrix.mul_sum, Matrix.sum_mul, Matrix.mul_smul, Matrix.smul_mul]
  have h2 : P (Lm c R' a) = ∑ κ, a κ • P (Lm c R' (Pi.single κ 1)) := by
    rw [Lm_eq_sum]
    induction (Finset.univ : Finset (Fin K)) using Finset.induction_on with
    | empty => rw [Finset.sum_empty, Finset.sum_empty, hP0]
    | insert x s hx ih => rw [Finset.sum_insert hx, Finset.sum_insert hx, hPadd, hPsmul, ih]
  rw [h1, h2]
  exact Finset.sum_congr rfl fun κ _ => by rw [hP]

theorem linearIndependent_int_of_uC (huind : LinearIndependent R (fun k : Fin K => uC u R k)) :
    LinearIndependent ℤ u := by
  rw [Fintype.linearIndependent_iff] at huind ⊢
  intro g hg k
  have h := congrArg (Int.castRingHom R).mapMatrix hg
  rw [map_sum, map_zero] at h
  have h' : ∑ i, (g i : R) • uC u R i = 0 := by
    rw [← h]
    refine Finset.sum_congr rfl fun i _ => ?_
    rw [map_zsmul, uC_eq, Int.cast_smul_eq_zsmul]
  exact (Int.castRingHom R).injective_int (by rw [map_zero]; exact huind _ h' k)

theorem u_comm_of_uC_comm (h : ∀ k l : Fin K, uC u R k * uC u R l = uC u R l * uC u R k) (k l : Fin K) :
    u k * u l = u l * u k := by
  have h1 : (Int.castRingHom R).mapMatrix (u k * u l) = (Int.castRingHom R).mapMatrix (u l * u k) := by
    rw [map_mul, map_mul]
    exact h k l
  exact Matrix.map_injective (RingHom.injective_int (Int.castRingHom R)) h1

theorem uC_comm_of_TC_comm (hmem : ∀ k, uC u R k ∈ Algebra.adjoin R (Set.range (TC T R)))
    (hT : ∀ g h : G, TC T R g * TC T R h = TC T R h * TC T R g) (k l : Fin K) :
    uC u R k * uC u R l = uC u R l * uC u R k := by
  have hgen : ∀ a ∈ Set.range (TC T R), ∀ b ∈ Set.range (TC T R), Commute a b := by
    rintro _ ⟨g, rfl⟩ _ ⟨h, rfl⟩
    exact hT g h
  have h1 : ∀ b ∈ Set.range (TC T R), Commute (uC u R k) b := fun b hb =>
    (Algebra.commute_of_mem_adjoin_of_forall_mem_commute (hmem k) fun a ha => hgen b hb a ha).symm
  exact (Algebra.commute_of_mem_adjoin_of_forall_mem_commute (hmem l) h1).eq
